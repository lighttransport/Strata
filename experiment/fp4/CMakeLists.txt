cmake_minimum_required(VERSION 3.24)
# Standalone MXFP4 decode/GEMM benchmarks for MiMo-V2.6-Flash. Not part of the Strata build.
#   cmake -S experiment/fp4 -B experiment/fp4/build -DCMAKE_CUDA_COMPILER=/usr/local/cuda/bin/nvcc
#   cmake --build experiment/fp4/build -j
project(fp4_bench LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(OpenMP REQUIRED)

add_executable(bench_cpu cpu/bench_cpu.cpp)
target_compile_options(bench_cpu PRIVATE -O3 -mavx2 -mfma -mf16c)
target_link_libraries(bench_cpu PRIVATE OpenMP::OpenMP_CXX)

add_executable(bench_exl3_cpu exl3/bench_exl3.cpp)
target_compile_features(bench_exl3_cpu PRIVATE cxx_std_20)
target_include_directories(bench_exl3_cpu PRIVATE ../../include)
target_compile_options(bench_exl3_cpu PRIVATE -O3 -mavx2 -mfma -mf16c)
target_link_libraries(bench_exl3_cpu PRIVATE OpenMP::OpenMP_CXX)

add_executable(bench_research_cpu research/bench.cpp)
target_compile_features(bench_research_cpu PRIVATE cxx_std_20)
target_include_directories(bench_research_cpu PRIVATE ../../include)
target_compile_options(bench_research_cpu PRIVATE -O3 -mavx2 -mfma -mf16c)
target_link_libraries(bench_research_cpu PRIVATE OpenMP::OpenMP_CXX)
add_executable(check_research_formats research/check_formats.cpp)
target_compile_features(check_research_formats PRIVATE cxx_std_20)
target_include_directories(check_research_formats PRIVATE ../../include)
target_compile_options(check_research_formats PRIVATE -O3 -mavx2 -mfma -mf16c)
target_link_libraries(check_research_formats PRIVATE OpenMP::OpenMP_CXX)

option(FP4_AMPERE "Build only Ampere-compatible GPU kernels (sm_86)" OFF)
option(FP4_CUDA "Build the GPU benchmark" ON)
if(FP4_CUDA)
  # sm_120a: the software paths only need sm_80 (int8 mma) / sm_89 (fp8 mma); the native
  # block-scaled FP4 instructions in gpu/mma_native.cu need the arch-specific 120a target.
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    if(FP4_AMPERE)
      set(CMAKE_CUDA_ARCHITECTURES 86)
    else()
      set(CMAKE_CUDA_ARCHITECTURES 120a)
    endif()
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  add_executable(bench_exl3_gpu exl3/bench_exl3.cpp ../../src/kernels/cuda/exl3.cu)
  target_compile_features(bench_exl3_gpu PRIVATE cxx_std_20)
  target_include_directories(bench_exl3_gpu PRIVATE ../../include)
  target_compile_definitions(bench_exl3_gpu PRIVATE EXL3_GPU=1)
  target_link_libraries(bench_exl3_gpu PRIVATE OpenMP::OpenMP_CXX CUDA::cudart)
  target_compile_options(bench_exl3_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3 -lineinfo>)
  add_executable(bench_research_gpu research/bench.cpp research/gemv.cu ../../src/kernels/cuda/exl3.cu)
  target_compile_features(bench_research_gpu PRIVATE cxx_std_20)
  target_include_directories(bench_research_gpu PRIVATE ../../include)
  target_compile_definitions(bench_research_gpu PRIVATE RESEARCH_GPU=1)
  target_link_libraries(bench_research_gpu PRIVATE OpenMP::OpenMP_CXX CUDA::cudart)
  target_compile_options(bench_research_gpu PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3 -mavx2 -mfma -mf16c> $<$<COMPILE_LANGUAGE:CUDA>:-O3 -lineinfo>)
  add_executable(bench_gpu gpu/bench_gpu.cu gpu/mma_native.cu gpu/mma_peak.cu)
  if(FP4_AMPERE)
    target_compile_definitions(bench_gpu PRIVATE FP4_AMPERE=1)
  endif()
  target_compile_options(bench_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3 -lineinfo --expt-relaxed-constexpr -Xcompiler=-fopenmp>)
  target_link_libraries(bench_gpu PRIVATE OpenMP::OpenMP_CXX)
endif()
