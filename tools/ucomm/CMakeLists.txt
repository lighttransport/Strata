cmake_minimum_required(VERSION 3.13)
project(ucomm C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(ucomm src/ucomm.c src/ucomm_boot.c src/ucomm_udp.c src/ucomm_ib.c src/ibvew.c src/ucomm_file.c src/ucomm_relay.c)
target_include_directories(ucomm PUBLIC include PRIVATE src)
target_link_libraries(ucomm PUBLIC ${CMAKE_DL_LIBS})
add_executable(ucomm_test_loopback tests/test_loopback.c)
target_link_libraries(ucomm_test_loopback ucomm)
add_executable(ucomm_bench tools/ucomm_bench.c)
target_link_libraries(ucomm_bench ucomm)
add_executable(ucfile tools/ucfile.c)
target_link_libraries(ucfile ucomm pthread)
add_executable(ucomm_test_file tests/test_file.c)
target_link_libraries(ucomm_test_file ucomm)
add_executable(ucomm_relay tools/ucomm_relay.c)
target_link_libraries(ucomm_relay ucomm)
add_executable(ucomm_test_relay_api tests/test_relay_api.c)
target_link_libraries(ucomm_test_relay_api ucomm)
add_executable(ucomm_test_progress tests/test_progress.c)
target_include_directories(ucomm_test_progress PRIVATE src)
target_link_libraries(ucomm_test_progress ucomm)

include(CTest)
if(BUILD_TESTING)
  add_test(NAME ucomm_progress COMMAND ucomm_test_progress)
  add_test(NAME ucomm_relay_api COMMAND ucomm_test_relay_api)
  add_test(NAME ucomm_relay_api_pair COMMAND ucomm_test_relay_api paired)
  set_tests_properties(ucomm_relay_api PROPERTIES TIMEOUT 25)
  set_tests_properties(ucomm_relay_api_pair PROPERTIES TIMEOUT 25)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME ucomm_relay_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_relay.py $<TARGET_FILE:ucomm_relay>)
    set_tests_properties(ucomm_relay_integration PROPERTIES TIMEOUT 180)
    add_test(NAME ucomm_relay_runner COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_relay_runner.py $<TARGET_FILE:ucomm_relay>)
    set_tests_properties(ucomm_relay_runner PROPERTIES TIMEOUT 60)
    add_test(NAME ucomm_bench_args COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bench.py $<TARGET_FILE:ucomm_bench>)
    set_tests_properties(ucomm_bench_args PROPERTIES TIMEOUT 25)
  endif()
endif()
